cmake_minimum_required(VERSION 3.22)
project(KestrelIsland VERSION 0.3.8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Widgets DBus Network Test)
find_package(LayerShellQt REQUIRED)
qt_standard_project_setup()
set(KESTREL_SOURCES src/backend.cpp src/backend.h src/systemmonitor.cpp src/systemmonitor.h src/activitycoordinator.cpp src/activitycoordinator.h
    src/focussession.cpp src/focussession.h src/keepawakeservice.cpp src/keepawakeservice.h
    src/taskactivityservice.cpp src/taskactivityservice.h src/islandcontrol.cpp src/islandcontrol.h
    src/sessionstate.cpp src/sessionstate.h src/notificationsservice.cpp src/notificationsservice.h
    src/languageservice.cpp src/languageservice.h)
qt_add_library(kestrel-core STATIC ${KESTREL_SOURCES})
target_link_libraries(kestrel-core PUBLIC Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::DBus Qt6::Network LayerShellQt::Interface)
target_include_directories(kestrel-core PUBLIC src)
qt_add_executable(kestrel-island src/main.cpp)
file(GLOB KESTREL_QML CONFIGURE_DEPENDS "qml/*.qml")
qt_add_resources(kestrel-island "ui" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${KESTREL_QML} qml/qmldir assets/island.svg)
file(GLOB ISLAND_ICONS CONFIGURE_DEPENDS "assets/icons/*.svg")
qt_add_resources(kestrel-island "icons" PREFIX "/assets/icons" BASE "assets/icons" FILES ${ISLAND_ICONS})
target_link_libraries(kestrel-island PRIVATE kestrel-core)
include(GNUInstallDirs)
install(TARGETS kestrel-island RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS scripts/kestrel-task DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.kestrel.Island.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/island.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME kestrel-island.svg)
include(CTest)
if(BUILD_TESTING)
    qt_add_executable(test-backend tests/test_backend.cpp)
    qt_add_resources(test-backend "test_ui" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${KESTREL_QML} qml/qmldir assets/island.svg)
    qt_add_resources(test-backend "test_icons" PREFIX "/assets/icons" BASE "assets/icons" FILES ${ISLAND_ICONS})
    target_include_directories(test-backend PRIVATE src)
    target_link_libraries(test-backend PRIVATE kestrel-core Qt6::Test)
    qt_add_executable(kestrel-desktop-probe tests/desktop_probe.cpp)
    target_link_libraries(kestrel-desktop-probe PRIVATE kestrel-core)
    qt_add_executable(test-activities tests/test_activities.cpp)
    target_link_libraries(test-activities PRIVATE kestrel-core Qt6::Test)
    find_program(DBUS_RUN_SESSION dbus-run-session)
    if(DBUS_RUN_SESSION)
        add_test(NAME activities COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:test-activities> -platform offscreen)
        add_test(NAME backend-kde-sliders COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:test-backend> pointerDragsReachMediaAndVolume expandedStaysUntilExplicitClose -platform offscreen)
        set_tests_properties(backend-kde-sliders PROPERTIES TIMEOUT 40 ENVIRONMENT "QT_QPA_PLATFORMTHEME=generic;QT_QUICK_CONTROLS_STYLE=org.kde.desktop;QT_QUICK_BACKEND=software")
        add_test(NAME backend COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:test-backend> -platform offscreen)
        find_package(Python3 COMPONENTS Interpreter REQUIRED)
        add_test(NAME cli COMMAND ${DBUS_RUN_SESSION} -- ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:kestrel-island>)
        set_tests_properties(cli activities backend PROPERTIES TIMEOUT 40 ENVIRONMENT "QT_QPA_PLATFORMTHEME=generic;QT_QUICK_CONTROLS_STYLE=Basic;QT_QUICK_BACKEND=software")
    else()
        message(WARNING "dbus-run-session missing: integration tests are not registered; never run them on the user bus")
    endif()
endif()
